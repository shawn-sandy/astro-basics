# API Reference

This section provides detailed API documentation for all components, utilities, and configuration options available in astro-basics.

## Component APIs

### Layout Components

#### BaseLayout

Main layout component for all pages.

**Props:**
```typescript
interface Props {
  title: string
  description?: string
  image?: string
  noindex?: boolean
  type?: 'website' | 'article'
}
```

**Usage:**
```astro
<BaseLayout 
  title="Page Title"
  description="Page description for SEO"
  image="/og-image.jpg"
  type="article"
/>
```

#### Header

Site navigation header component.

**Props:**
```typescript
interface Props {
  title?: string
  showAuth?: boolean
  theme?: 'light' | 'dark' | 'auto'
  sticky?: boolean
}
```

### Content Components

#### PostCard

Card component for displaying blog posts and articles.

**Props:**
```typescript
interface Props {
  title: string
  description: string
  pubDate: Date
  author: string
  href: string
  featured?: boolean
  image?: {
    url: string
    alt: string
  }
  tags?: string[]
}
```

**Example:**
```astro
<PostCard 
  title="Getting Started with Astro"
  description="Learn the basics of Astro development"
  pubDate={new Date('2025-01-15')}
  author="John Doe"
  href="/posts/getting-started"
  featured={true}
  tags={['astro', 'tutorial']}
/>
```

## Utility APIs

### Content Utilities

#### getCollection Helper

Enhanced content collection utilities.

```typescript
import { getCollection } from 'astro:content'

// Get published posts only
const publishedPosts = await getCollection('posts', ({ data }) => data.publish)

// Get featured content
const featuredPosts = await getCollection('posts', ({ data }) => 
  data.publish && data.featured
)
```

#### Slugify Function

URL-safe slug generation.

```typescript
import { slugify } from '#libs/content'

const slug = slugify("My Blog Post Title") // "my-blog-post-title"
```

#### Truncate Function

Text truncation with word boundaries.

```typescript
import { truncate } from '#libs/content'

const shortText = truncate("Long text content...", 100) // Truncated to 100 chars
```

### Authentication APIs

#### Clerk Integration

Pre-configured authentication with Clerk.

```typescript
// In API routes
import { getAuth } from '@clerk/astro/server'

export async function GET({ request }) {
  const { userId } = getAuth(request)
  
  if (!userId) {
    return new Response('Unauthorized', { status: 401 })
  }
  
  // Authenticated route logic
}
```

#### Protected Routes

Middleware for route protection.

```typescript
// src/middleware.ts
import { clerkMiddleware } from '@clerk/astro/server'

export const onRequest = clerkMiddleware()
```

### Content Schema

**File:** `src/content/config.ts`

```typescript
const postsCollection = defineCollection({
  schema: z.object({
    title: z.string(),
    pubDate: z.date(),
    description: z.string(),
    author: z.string(),
    breadcrumbSlug: z.string().optional(),
    image: z.object({
      url: z.string(),
      alt: z.string(),

    featured: z.boolean().default(false),
    youtube: z.object({
      id: z.string(),

      start: z.string().optional(),
      end: z.string().optional(),
    }).optional(),
  }),
})
```

## Error Handling

### API Route Patterns

```typescript
export async function GET({ request }) {
  try {
    // Route logic
    return new Response(JSON.stringify({ success: true, data }), {
      headers: { 'Content-Type': 'application/json' }
    })
  } catch (error) {
    console.error('API Error:', error)
    return new Response(JSON.stringify({ 
      success: false, 
      error: error.message 
    }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    })
  }
}
```

### Component Error Boundaries

```astro
---
let errorMessage: string | null = null

try {
  // Component logic that might throw
} catch (error) {
  errorMessage = error.message
}
---

{errorMessage ? (
  <div class="error-message">
    <p>Error: {errorMessage}</p>
  </div>
) : (
  <!-- Normal component content -->
)}
```
